<script>
(() => {
  'use strict';

const $ = id => document.getElementById(id);
  const canvas = $('map');
  const ctx = canvas.getContext('2d');
  const COLS = 10, ROWS = 7, STEP_X = 98, STEP_Y = 88;
  const WORLD_W = 1000, WORLD_H = 650;
  const KM_PER_UNIT = 0.002;
  const PLAYBACK = 20;

const places = [
    { id: 61, name: 'Westside charging hub' },
    { id: 18, name: 'Innovation campus' },
    { id: 23, name: 'Central park entrance' },
    { id: 46, name: 'Greenfield station' },
    { id: 5, name: 'North market' },
    { id: 68, name: 'Riverside terminal' }
  ];

for (const place of places) {
    for (const target of ['start', 'end']) {
      const option = document.createElement('option');
      option.value = place.id;
      option.textContent = place.name;
      $(target).appendChild(option);
    }
  }
  $('start').value = 61;
  $('end').value = 18;

const state = {
    path: [], lengths: [], total: 0, traveled: 0, elapsed: 0,
    running: false, complete: false, weather: 'clear', layer: 'map',
    closures: new Set(), start: 61, end: 18
  };
  let viewport = { scale: 1, x: 0, y: 0, width: 1, height: 1 };
  let toastTimer;
  const point = id => ({ x: 60 + (id % COLS) * STEP_X, y: 60 + Math.floor(id / COLS) * STEP_Y });
  const edgeLength = (a, b) => {
    const p = point(a), q = point(b);
    return Math.hypot(p.x - q.x, p.y - q.y);
  };
  const congestion = (a, b) => {
    const seed = Math.min(a, b) * 71 + Math.max(a, b) * 37;
    return (Math.sin(seed) + 1) / 2;
  };
  const weatherFactor = () => ({ clear: 1, rain: .75, fog: .55 })[state.weather];
  const edgeSpeed = (a, b) => +$('speed').value * weatherFactor()
    * (1 - (+$('traffic').value / 100) * .65 * congestion(a, b));

function neighbors(id) {
    const x = id % COLS, y = Math.floor(id / COLS), result = [];
    if (x > 0) result.push(id - 1);
    if (x < COLS - 1) result.push(id + 1);
    if (y > 0) result.push(id - COLS);
    if (y < ROWS - 1) result.push(id + COLS);
    return result.filter(n => !state.closures.has(n));
  }

// Dijkstra routing minimizes estimated travel time through the street graph.
  function calculateRoute() {
    const distances = Array(COLS * ROWS).fill(Infinity);
    const previous = Array(COLS * ROWS).fill(-1);
    const visited = new Set();
    distances[state.start] = 0;

while (visited.size < COLS * ROWS) {
      let current = -1;
      for (let i = 0; i < distances.length; i++) {
        if (!visited.has(i) && (current < 0 || distances[i] < distances[current])) current = i;
      }
      if (current < 0 || !Number.isFinite(distances[current])) break;
      if (current === state.end) break;
      visited.add(current);

for (const next of neighbors(current)) {
        const cost = distances[current] + edgeLength(current, next) / edgeSpeed(current, next);
        if (cost < distances[next]) {
          distances[next] = cost;
          previous[next] = current;
        }
      }
    }

if (!Number.isFinite(distances[state.end])) return [];
    const path = [];
    for (let node = state.end; node !== -1; node = previous[node]) path.unshift(node);
    return path;
  }

function notify(message) {
    $('toast').textContent = message;
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => $('toast').textContent = '', 4200);
  }

function formatTime(seconds) {
    const whole = Math.max(0, Math.round(seconds));
    return `${String(Math.floor(whole / 60)).padStart(2, '0')}:${String(whole % 60).padStart(2, '0')}`;
  }

function plan() {
    state.start = +$('start').value;
    state.end = +$('end').value;
    state.closures.delete(state.start);
    state.closures.delete(state.end);
    state.running = false;
    state.complete = false;
    state.traveled = 0;
    state.elapsed = 0;
    state.path = calculateRoute();
    state.lengths = state.path.slice(1).map((n, i) => edgeLength(state.path[i], n));
    state.total = state.lengths.reduce((sum, d) => sum + d, 0);
    if (state.start === state.end) state.complete = true;
    updateUI();
  }

function currentPosition() {
    if (!state.path.length) return { ...point(state.start), angle: 0, segment: 0 };
    let remaining = state.traveled;
    for (let i = 0; i < state.lengths.length; i++) {
      const length = state.lengths[i];
      if (remaining < length || i === state.lengths.length - 1) {
        const a = point(state.path[i]), b = point(state.path[i + 1]);
        const t = Math.min(1, remaining / length);
        return {
          x: a.x + (b.x - a.x) * t,
          y: a.y + (b.y - a.y) * t,
          angle: Math.atan2(b.y - a.y, b.x - a.x), segment: i
        };
      }
      remaining -= length;
    }
    return { ...point(state.end), angle: 0, segment: 0 };
  }

function updateUI() {
    const fraction = state.total ? state.traveled / state.total : (state.complete ? 1 : 0);
    const position = currentPosition();
    const speed = state.running && state.path.length > 1
      ? edgeSpeed(state.path[position.segment], state.path[position.segment + 1]) : 0;
    const eta = state.lengths.reduce((sum, length, i) =>
      sum + length * KM_PER_UNIT / edgeSpeed(state.path[i], state.path[i + 1]) * 3600, 0);

$('distance').textContent = (state.total * KM_PER_UNIT).toFixed(2);
    $('eta').textContent = state.path.length ? formatTime(eta) : '—';
    $('currentSpeed').textContent = Math.round(speed);
    $('closureCount').textContent = state.closures.size;
    $('completion').textContent = Math.round(fraction * 100);
    $('traveled').textContent = (state.traveled * KM_PER_UNIT).toFixed(2);
    $('elapsed').textContent = formatTime(state.elapsed);
    $('progress').style.width = $('distanceProgress').style.width = `${fraction * 100}%`;
    $('speedLabel').textContent = `${$('speed').value} km/h`;
    const traffic = +$('traffic').value;
    $('trafficLabel').textContent = traffic < 25 ? 'Light' : traffic < 65 ? 'Moderate' : 'Heavy';
    $('run').textContent = state.running ? 'Ⅱ  Pause simulation'
      : state.complete ? '↻  Run again'
      : state.traveled ? '▶  Resume simulation' : '▶  Start simulation';
    $('driveState').textContent = state.running ? 'Autonomous driving'
      : state.complete ? 'Destination reached' : state.traveled ? 'Paused' : 'Ready to start';
    $('status').textContent = !state.path.length ? 'No route available. Remove a road closure.'
      : state.start === state.end ? 'Start and destination are the same.'
      : state.complete ? 'Journey complete. Destination reached.'
      : state.running ? 'Navigating the planned route…'
      : state.traveled ? 'Simulation paused.' : 'Route calculated. Ready to simulate.';
  }

function roundedRect(x, y, w, h, radius, fill, stroke) {
    ctx.beginPath();
    ctx.roundRect(x, y, w, h, radius);
    if (fill) { ctx.fillStyle = fill; ctx.fill(); }
    if (stroke) { ctx.strokeStyle = stroke; ctx.lineWidth = 1; ctx.stroke(); }
  }

function line(a, b, color, width) {
    ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y);
    ctx.strokeStyle = color; ctx.lineWidth = width; ctx.stroke();
  }

function circle(x, y, radius, color) {
    ctx.beginPath(); ctx.arc(x, y, radius, 0, Math.PI * 2);
    ctx.fillStyle = color; ctx.fill();
  }

function resize() {
    const bounds = canvas.getBoundingClientRect();
    const dpr = window.devicePixelRatio || 1;
    canvas.width = Math.round(bounds.width * dpr);
    canvas.height = Math.round(bounds.height * dpr);
    const scale = Math.min(bounds.width / WORLD_W, bounds.height / WORLD_H);
    viewport = {
      width: bounds.width, height: bounds.height, scale,
      x: (bounds.width - WORLD_W * scale) / 2,
      y: (bounds.height - WORLD_H * scale) / 2
    };
  }
  new ResizeObserver(resize).observe(canvas);

  });

let lastFrame = 0, lastUI = 0;
  function frame(now) {
    const dt = lastFrame ? Math.min((now - lastFrame) / 1000, .1) : 0;
    lastFrame = now;

if (state.running && state.total > 0) {
      let remainingTime = dt * PLAYBACK;
      // Consume time per segment so traffic-dependent speeds stay consistent.
      while (remainingTime > 0 && state.traveled < state.total) {
        const car = currentPosition();
        const segment = car.segment;
        const boundary = state.lengths.slice(0, segment + 1).reduce((sum, n) => sum + n, 0);
        const unitsPerSecond = edgeSpeed(state.path[segment], state.path[segment + 1]) / 3600 / KM_PER_UNIT;
        const timeToBoundary = Math.max(0, boundary - state.traveled) / unitsPerSecond;
        const used = Math.min(remainingTime, timeToBoundary);
        state.traveled = Math.min(state.total, state.traveled + used * unitsPerSecond);
        state.elapsed += used;
        remainingTime -= used;
        if (timeToBoundary <= used + 1e-9) state.traveled = Math.min(state.total, boundary + 1e-8);
        if (used === 0 && timeToBoundary > 0) break;
      }
      if (state.traveled >= state.total) {
        state.traveled = state.total;
        state.running = false;
        state.complete = true;
        notify('Destination reached. Your simulation is complete.');
        updateUI();
      }
    }

drawMap(now);
    if (now - lastUI > 120) { updateUI(); lastUI = now; }
    requestAnimationFrame(frame);
  }

plan();
  resize();
  requestAnimationFrame(frame);
})();
